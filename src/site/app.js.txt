function approx(ad) {
  var cur = ad.cc && D.currency && D.currency[ad.cc];
  if (!cur || !ad.p || !D.fx || !D.fx[cur[0]]) return '';
  var r = D.fx[cur[0]];

function filtered() {
  var q = state.q.trim().toLowerCase();
  var list = (D.ads || []).filter(function (a) {
    return (state.cat === 'all' || a.cat === state.cat) && (state.country === 'all' || a.country === state.country) &&
      (!state.city || a.city === state.city) && (!state.tag || (a.tags || []).some(function (x) { return x.toLowerCase() === state.tag.toLowerCase(); })) &&
      (!q || (a.title + ' ' + a.text + ' ' + a.city + ' ' + (a.tags || []).join(' ').replace(/_/g, ' ')).toLowerCase().indexOf(q) >= 0);
  });
  if (q) list.forEach(function (a) { a._tagHit = (a.tags || []).some(function (x) { return x.replace(/_/g, ' ').toLowerCase().indexOf(q) >= 0; }) ? 1 : 0; });
  if (state.sort === 'cheap') list.sort(function (a, b) { return (a.p || 1e12) - (b.p || 1e12); });
  else if (state.sort === 'dear') list.sort(function (a, b) { return b.p - a.p; });
  else list.sort(function (a, b) { return (b.ts + Math.min(b.boost, 30) * 3600) - (a.ts + Math.min(a.boost, 30) * 3600); });
  if (mycity) list.sort(function (a, b) { return (b.city === mycity) - (a.city === mycity); });
  if (q) list.sort(function (a, b) { return (b._tagHit || 0) - (a._tagHit || 0); });
  return list;
}

function submitPost() {
  var form = $('post-form'), fd = new FormData(form), err = $('f-err');
  fd.set('mode', state.mode);
  var cityPick = $('f-city').value;
  fd.set('city', cityPick === '__other' ? $('f-city-other').value.trim() : cityPick);
  fd.set('tg', state.contacts.tg ? '1' : '0');
  fd.set('site', form.querySelector('[name=site]').checked ? '1' : '0');
  if (state.edit) fd.set('replace', state.edit.id);
  var text = (fd.get('text') || '').trim();
  if (state.mode === 'form') {
    if (!(fd.get('title') || '').trim() || !text || !(state.contacts.tg || state.contacts.ph) || !fd.get('city')) { err.textContent = tr('f_need'); err.hidden = false; return; }
    if (state.contacts.tg && !state.me.nick && !(fd.get('phone') || '').trim()) { err.textContent = tr('f_nonick'); err.hidden = false; return; }
  } else if (text.length < 10) { err.textContent = tr('f_need'); err.hidden = false; return; }
  var btn = $('send'); btn.disabled = true; btn.textContent = tr('sending');
  api('/api/ads', { form: fd }).then(function (r) {
    btn.disabled = false; btn.textContent = tr('send');
    if (r && r.ok) { state.published = r.published; state.step = 3; state.edit = null; refreshMe(); drawPost(); return; }
    var e = r && r.error;
    if (e === 'membership') { state.member = r.member; state.step = 1; drawPost(); return; }
    if (e === 'auth') { state.me = null; state.step = 0; renderAuth(); drawPost(); return; }
    err.textContent = e === 'phone' ? tr('err_phone') : e === 'photo_big' ? tr('err_big') : e === 'limit_day' ? tr('limit_none', { n: Math.max(1, Math.ceil(((r.nextAt || 0) * 1000 - Date.now()) / 3600000)) })
      : e === 'limit_active' ? tr('limit_active', { n: r.active }) : e === 'text' || e === 'contact' || e === 'fields' ? tr('f_need') : tr('err');
    err.hidden = false;
  });
}

function statusBadge(a) {
  if (a.status === 'published') return '<span class="status s-pub">' + tr('s_pub', { n: a.daysLeft }) + '</span>';
  if (a.status === 'pending') return '<span class="status s-mod">' + tr('s_mod') + '</span>';
  if (a.status === 'rejected') return '<span class="status s-rej">' + tr('s_rej') + '</span>';
  return '<span class="status s-exp">' + tr('s_exp') + '</span>';
}
function myActions(a) {
  var b = function (act, label) { return '<button class="ghost" type="button" data-my="' + act + ':' + a.id + '">' + label + '</button>'; };
  if (a.status === 'published') return '<a class="ghost" href="/ad/' + a.id + '">' + tr('a_open') + '</a>' + b('ext', tr('a_ext')) + b('edit', tr('a_edit')) + b('del', tr('a_del'));
  if (a.status === 'pending') return b('del', tr('a_del'));
  if (a.status === 'rejected') return b('edit', tr('a_edit')) + b('del', tr('a_del'));

}

        return '<div class="adm-card"><p class="sm"><b>Скарга #' + x.id + '</b> · ' + esc(x.reason) + ' · скарг: ' + x.count + ' · ' + fmtTs(x.created) + '</p>' +
          (x.ad ? '<h4><a href="#" data-admopen="' + x.ad.id + '">' + x.ad.no + '</a> ' + esc(x.ad.title) + ' <span class="sm">(' + (stLabel[x.ad.status] || x.ad.status) + ')</span></h4>' : '') +
          (x.target ? '<p class="sm">На кого: ' + who(x.target) + '</p>' : '') + (x.note ? '<pre>' + esc(x.note) + '</pre>' : '') + (x.msgText ? '<pre>' + esc(x.msgText) + '</pre>' : '') +
          (x.msgLink ? '<p class="sm"><a href="' + esc(x.msgLink) + '" target="_blank" rel="noopener">Повідомлення в чаті</a></p>' : '') +
          '<div class="adm-acts">' + b('rdismiss', x.id, '✅ Відхилити скаргу') + (x.ad && x.ad.status !== 'deleted' ? b('rdelad', x.id, '🗑 Видалити оголошення') : '') + (x.target ? b('rban', x.id, '⛔ Бан + видалити оголошення') : '') + '</div></div>';
      }).join('') : '<p class="sub">✨ Відкритих скарг немає.</p>';
    });
    if (tab === 'find') {
      body.innerHTML = '<form class="adm-find" id="adm-find"><input id="adm-no" placeholder="#00000001" autocomplete="off" inputmode="text"><button class="btn" type="submit">Знайти</button></form><div id="adm-found"></div>';
      $('adm-find').onsubmit = function (e) {
        e.preventDefault();
        var v = $('adm-no').value.trim();
        if (!/^#\d{8}$/.test(v)) { $('adm-found').innerHTML = '<p class="sub">Введіть повний номер з решіткою, наприклад #00000012.</p>'; return; }
        openAd(Number(v.slice(1)));
      };
      if (D.adId) { $('adm-no').value = '#' + String(D.adId).padStart(8, '0'); openAd(D.adId); D.adId = null; }
    }
    if (tab === 'stats') api('/api/admin/stats').then(function (r) {
      if (!r || !r.ok) return denied();
      var s = r.stats, w = s.week;
      var box = function (n, l) { return '<div><b>' + (n == null ? '—' : n) + '</b><span>' + l + '</span></div>'; };
      body.innerHTML = '<div class="adm-stats">' + box(s.channel, 'підписників каналу') + box(s.group, 'учасників групи') + box(s.users, 'користувачів бота') +
        box(s.active, 'активних оголошень') + box(s.pending, 'на модерації') + box(s.reports, 'відкритих скарг') + '</div><h3 style="margin:22px 0 10px">За 7 днів</h3><div class="adm-stats">' +
        box(w.ads, 'подано оголошень') + box(w.published, 'опубліковано') + box(w.rejected, 'відхилено') + box(w.newUsers, 'нових користувачів') +
        box(w.siteViews, 'відвідувань сайту') + box(w.adViews, 'переглядів оголошень') + box(w.contacts, 'відкрили контакт') + box(w.siteAds, 'оголошень із сайту') + box(w.logins, 'входів на сайт') + '</div>';
    });
  }
  function openAd(id) {
    var box = $('adm-found') || body;
    box.innerHTML = '<p class="sub">Шукаю…</p>';
    api('/api/admin/ad/' + id).then(function (r) {
      if (r && r.error === 'forbidden') return denied();
      box.innerHTML = r && r.ok ? adCard(r.ad, fullActs) : '<p class="sub">Оголошення #' + String(id).padStart(8, '0') + ' не знайдено.</p>';
    });
  }
  function denied() { body.innerHTML = '<p class="sub">Немає доступу. Увійдіть через Telegram з акаунта власника.</p>'; }
  document.addEventListener('click', function (e) {
    var t = e.target;
    var tb = t.closest('[data-admtab]');
    if (tb) { setTab(tb.getAttribute('data-admtab')); return; }
    var op = t.closest('[data-admopen]');
    if (op) { e.preventDefault(); setTab('find'); $('adm-no').value = '#' + String(op.getAttribute('data-admopen')).padStart(8, '0'); openAd(Number(op.getAttribute('data-admopen'))); return; }
    var a = t.closest('[data-adm]');
    if (!a) return;
    var p = a.getAttribute('data-adm').split(':'), act = p[0], id = p[1], url, payload = {};
    if (act === 'delete' && !confirm('Видалити оголошення з каналу і сайту?')) return;
    if ((act === 'ban' || act === 'rban') && !confirm('Заблокувати автора назавжди і видалити всі його оголошення?')) return;
    if (act.charAt(0) === 'r' && act !== 'reject') url = '/api/admin/report/' + id + '/' + act.slice(1);
    else url = '/api/admin/ad/' + id + '/' + act;
    if (act === 'reject') { var sel = document.querySelector('[data-reason="' + id + '"]'); payload.reason = sel ? sel.value : 'rules'; }
    a.disabled = true;
    api(url, { body: payload }).then(function (r) {
      a.disabled = false;
      toast(r && r.ok ? (r.msg || 'Готово') : 'Помилка' + (r && r.msg ? ': ' + r.msg : r && r.error ? ': ' + r.error : ''));
      counts();
      if (tab === 'find' && $('adm-no')) openAd(Number(id)); else draw();
    });
  });
  setTab(tab);
  counts();
})();

renderAll();
refreshMe();
